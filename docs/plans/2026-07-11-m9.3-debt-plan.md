- [ ] **Step 3: 實作 withBacklogLock**

`src/backlog.ts` 加（import `mkdirSync, rmdirSync, statSync` 等併入檔頭既有 fs import）：

```ts
/**
 * backlog 檔級跨進程互斥鎖：mkdirSync 原子性當 mutex。
 * 等待上限 5s（10ms 間隔輪詢）；鎖目錄 mtime 超過 10s 視為殘留（持鎖進程死亡）強拆。
 * bot(appendUserTask) 與 daemon(BacklogStore.append/report) 都必須經此函式寫 backlog，
 * 否則 M8.1 記載的 lost-update 殘窗（read 與 write 兩 syscall 之間被整檔重寫夾擊）依然存在。
 */
export function withBacklogLock<T>(file: string, fn: () => T): T {
  const dir = `${file}.lockdir`
  const deadline = Date.now() + 5000
  const buf = new Int32Array(new SharedArrayBuffer(4))
  for (;;) {
    try { mkdirSync(dir); break } catch {
      try { if (Date.now() - statSync(dir).mtimeMs > 10_000) { rmdirSync(dir); continue } } catch { continue }
      if (Date.now() > deadline) throw new Error(`backlog 鎖等待逾時：${dir}`)
      Atomics.wait(buf, 0, 0, 10)
    }
  }
  try { return fn() } finally { try { rmdirSync(dir) } catch { /* 已被 stale 強拆屬可容忍 */ } }
}
```

`BacklogStore.append` 與 `report` 的整個函式體各包一層 `return withBacklogLock(this.file, () => { ...原函式體... })`。

function loadOrCreateToken(dataDir) {
